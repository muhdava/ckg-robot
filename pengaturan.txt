# Berkas CSV berisi data peserta. Cukup nama berkasnya bila berada di

# Tahap yang dijalankan:
#   penuh       = daftar -> konfirmasi hadir -> mulai -> isi semua form
#                 -> selesaikan layanan          (dipakai sehari-hari)
#   pendaftaran = mendaftar saja
#   konfirmasi  = peserta sudah punya tiket, lanjut konfirmasi + mulai
#   isi         = peserta sudah "Sedang Pemeriksaan", isi formnya saja

# Kategori peserta:
#   auto     = ikuti isi berkas CSV-nya (DISARANKAN)
#   dewasa   = paksa alur individu (Dewasa/Lansia)
#   sekolah  = paksa alur murid (CKG Sekolah)
# Biarkan "auto" saja: berkas sekolah otomatis memakai alur sekolah, jadi
# tidak mungkin salah pasang.
kategori = auto


# Batas bawaan bila saat ditanya "Nomor data" kamu langsung menekan Enter.
# Sehari-hari tidak perlu diubah — nomor data ditanyakan setelah login.

# ---------------------------------------------------------------------
#  Pengaman

# ya  = jalankan semua langkah TAPI berhenti sebelum mendaftarkan,
#       sehingga tidak ada data peserta yang tercatat di portal.
# tidak = pendaftaran sungguhan.

# Tekan "Selesaikan Layanan" setelah semua form target terisi.
# Tombol itu di portal selalu aktif walau masih ada pemeriksaan kosong,
# jadi program hanya menekannya bila pembacaan status tiap baris
# menyatakan semuanya sudah selesai.
selesaikan layanan = ya

# Lewati peserta yang pada run sebelumnya sudah tercatat selesai.
# Inilah yang membuat batch panjang aman dijalankan ulang setelah putus:
# tinggal jalankan lagi dengan rentang yang sama.
lewati sudah selesai = ya

# Jalankan lagi peserta yang sebelumnya DITOLAK portal (tidak valid).
# Biarkan 'tidak': portal akan menolak data yang sama dengan kalimat yang
# sama. Ubah ke 'ya' hanya sesudah datanya dibetulkan (NIK tetap sama).
ulang tidak valid = tidak


# ---------------------------------------------------------------------

# Batas berapa browser boleh dibuka sekaligus - itulah pilihan yang muncul di
# kotak "Jumlah pekerja" pada halaman aplikasi.
#
# TIAP BROWSER BUTUH AKUN PUSKESMAS SENDIRI yang kamu login-kan satu per satu.
# Jadi isi paling banyak sebanyak akun yang benar-benar kamu punya; menaikkan
# angka ini tidak menambah akun. Tiap browser tambahan juga memakan memori dan
# menambah beban ke portal.

# Berapa kali daftar layanan disapu ulang untuk mengejar form yang gagal.

# Berapa kali halaman daftar layanan dimuat ulang bila isinya tidak ter-render.

# Batas jumlah form per peserta; 0 = tanpa batas.
# Isi 1 bila ingin mencoba satu form dulu.

# tidak = browser berjalan tersembunyi. Biarkan "ya" supaya bisa diawasi,
# dan karena login harus kamu lakukan sendiri di jendela itu.

# Hentikan batch bila sekian peserta gagal BERTURUT-TURUT.
# Itu gejala khas browser yang sudah melambat: langkah yang tadinya
# lancar mulai gagal terus. Tutup browser, jalankan lagi dengan rentang
# yang sama - yang sudah selesai dilewati sendiri. 0 = jangan berhenti.

# Dipakai bila kolom No HP peserta kosong.

# Berapa lama menunggu nomor tiket terbit sesudah "Daftarkan dengan NIK" (ms).

# Berapa lama menunggu peserta yang baru didaftarkan muncul di daftar (ms).

# CKG Sekolah: tekan "Cek NIK" saat mendaftarkan murid, atau ketik seluruh

# Ini TIDAK berlaku untuk Dewasa/Lansia - di sana Cek NIK tetap dipakai.

# Berapa kali satu peserta diulang bila yang bermasalah ternyata PORTALNYA
